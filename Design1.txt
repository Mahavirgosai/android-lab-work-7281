package com.example.myapplication;

import android.content.Intent;
import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    EditText e_name,e_contect;
    Button b_design1;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);



        b_design1 = findViewById(R.id.b_design1);
        e_name = findViewById(R.id.m_e_name);
        e_contect = findViewById(R.id.m_e_contact);



        b_design1.setOnClickListener(v-> {
            String name = e_name.getText().toString();
            String contact = e_contect.getText().toString();

            Intent i = new Intent(MainActivity.this,design1.class);
            i.putExtra("name",name);
            i.putExtra("contect",contact);
            startActivity(i);
        });
    }
}
